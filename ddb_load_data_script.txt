# ## 2. create external table with same schema as your data - store it as Optimized Row Columnar (ORC)
sh> hive
hive> CREATE EXTERNAL TABLE oft_temp (hive_oid string, hive_si string, hive_p string, hive_d string, hive_ttl bigint) STORED AS ORC;

*ctrl+c*

# ## 3. read your csv data into a pyspark dataframe and create a View and insert it into the Hive table created above
sh>pyspark
pyspark> order = spark.read.csv('s3://neptune-bulkload-input/20211111/schema2/ddb_oid_features_load.csv/part-00000-0f4e1efc-0697-473a-a5c6-1abcaaa66fd1-c000.csv', header=True)
pyspark> order.createOrReplaceTempView("df")
pyspark> spark.sql("INSERT OVERWRITE TABLE oft_temp SELECT * FROM df")

*ctrl+d*

# ## 4. check whether the data has been loaded
sh> hive
hive> select * from oft_temp limit 5;

# ## 5. create the DynamoDB (DDB) connector table which defines column mapping
hive> CREATE EXTERNAL TABLE hive_ddb_conn (hive_oid string, hive_si string, hive_p string, hive_d string, hive_ttl bigint)
    > STORED BY 'org.apache.hadoop.hive.dynamodb.DynamoDBStorageHandler' 
    > TBLPROPERTIES ("dynamodb.table.name" = "DataStores-Devo-<region>-OrderFeaturesTable", 
    > "dynamodb.column.mapping" = "hive_oid:OID,hive_si:SI,hive_P:P,hive_d:D,hive_ttl:ttl");
    
# ## 6. overwrite the DDB connector table with data from CSV which we had loaded into hive table
hive> INSERT OVERWRITE TABLE hive_ddb_conn SELECT * FROM oft_temp;
